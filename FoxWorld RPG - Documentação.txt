1. 🎯 Visão Geral do Projeto

FoxWorld RPG é um jogo de RPG 2D por turnos desenvolvido em Python com Pygame, atualmente em fase de desenvolvimento inicial. O projeto possui uma base sólida de sistemas de interface e configurações.

Estado Atual: Fase 1 - Sistemas de menu e configurações completos
Próximo Foco: Implementação da jogabilidade principal

Características Principais:

✅ Sistema de estados robusto com State Pattern

✅ UI responsiva que se adapta a qualquer resolução

✅ Sistema de temas dinâmico e extensível

✅ Persistência de dados em SQLite com migração automática

✅ Sistema de configurações centralizado

✅ Criação de personagem completa com 5 classes

class StateManager:
    def push_state(self, state):    # Adiciona estado à pilha
    def pop_state(self):            # Remove estado atual  
    def change_state(self, state):  # Substitui estado atual
    def handle_event(self, event):  # Delega evento → estado.handle_event(event)
    def update(self):               # Atualiza → estado.update()
    def render(self, surface):      # Renderiza → estado.render(surface)
    def on_resize(old_size, new_size): # Notifica estados sobre redimensionamento

CREATE TABLE save_games (

    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    player_data TEXT NOT NULL,           -- Dados do jogador (JSON)
    world_state TEXT NOT NULL,           -- Estado do mundo (JSON)

)

import pygame
from src.states.base_state import BaseState
from src.ui.button import Button
from src.ui.button_manager import ButtonManager

class InventoryState(BaseState):
    def __init__(self, game):
        super().__init__(game)
        self.buttons = []
        self.items = []
        self._create_ui()
    
    def _create_ui(self):
        # Botão de voltar
        self.buttons.append(Button(50, 50, 200, 60, "Voltar", self._voltar, 28))
        
        # Botões de itens (exemplo)
        for i in range(5):
            self.buttons.append(Button(
                400, 150 + i * 80, 300, 60,
                f"Item {i+1}", 
                lambda idx=i: self._usar_item(idx),
                24
            ))
    
    def _voltar(self):
        self.game.state_manager.pop_state()
    
    def _usar_item(self, index):
        print(f"Usando item {index}")
        # Lógica de uso do item
    
    def handle_event(self, event):
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE or event.key == pygame.K_i:
                self._voltar()
        
        ButtonManager.handle_button_click(self.buttons, event)
    
    def update(self):
        ButtonManager.update_buttons(self.buttons, self.game)
    
    def render(self, surface):
        # Fundo semi-transparente
        s = pygame.Surface(surface.get_size(), pygame.SRCALPHA)
        s.fill((0, 0, 0, 200))  # Preto semi-transparente
        surface.blit(s, (0, 0))
        
        # Título
        title_font = self.game.game_config.get_font('title', 48)
        title_text = title_font.render("Inventário", True, (255, 255, 255))
        surface.blit(title_text, (400, 80))
        
        # Renderizar botões
        ButtonManager.render_buttons(self.buttons, surface, self.game.game_config)

8.2. Exemplo: Sistema de Diálogo com NPC

python
class DialogueState(BaseState):
    def __init__(self, game, npc_name, dialogue_lines):
        super().__init__(game)
        self.npc_name = npc_name
        self.dialogue_lines = dialogue_lines
        self.current_line = 0
        self.buttons = []
        self._create_ui()
    
    def _create_ui(self):
        self.buttons.append(Button(700, 600, 200, 50, "Próximo", self._proxima_linha, 24))
        self.buttons.append(Button(700, 670, 200, 50, "Sair", self._sair, 24))
    
    def _proxima_linha(self):
        self.current_line += 1
        if self.current_line >= len(self.dialogue_lines):
            self._sair()
    
    def _sair(self):
        self.game.state_manager.pop_state()
    
    def handle_event(self, event):
        ButtonManager.handle_button_click(self.buttons, event)
    
    def update(self):
        ButtonManager.update_buttons(self.buttons, self.game)
    
    def render(self, surface):
        # Fundo semi-transparente
        s = pygame.Surface(surface.get_size(), pygame.SRCALPHA)
        s.fill((0, 0, 0, 150))
        surface.blit(s, (0, 0))
        
        # Caixa de diálogo
        dialog_rect = pygame.Rect(200, 400, 1000, 200)
        pygame.draw.rect(surface, (40, 40, 60), dialog_rect, border_radius=10)
        pygame.draw.rect(surface, (80, 80, 100), dialog_rect, 3, border_radius=10)
        
        # Nome do NPC
        name_font = self.game.game_config.get_font('title', 32)
        name_text = name_font.render(self.npc_name, True, (255, 200, 100))
        surface.blit(name_text, (220, 360))
        
        # Texto do diálogo
        if self.current_line < len(self.dialogue_lines):
            dialog_font = self.game.game_config.get_font('menu', 24)
            dialog_text = dialog_font.render(
                self.dialogue_lines[self.current_line], 
                True, (255, 255, 255)
            )
            surface.blit(dialog_text, (230, 430))
        
        # Botões
        ButtonManager.render_buttons(self.buttons, surface, self.game.game_config)
